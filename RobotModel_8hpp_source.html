333 static RobotModel *createInstance(const std::string& name) {
334 RobotModelMap::iterator it = getRobotModelMap()->find(name);
335 if(it == getRobotModelMap()->end())
336 throw std::runtime_error("Failed to create instance of plugin " + name + ". Is the plugin registered?");
337 return it->second();
338 }
339
340 template<typename T>
341 static T* createInstance(const std::string& name){
342 RobotModel* tmp = createInstance(name);
343 T* ret = dynamic_cast<T*>(tmp);
344 return ret;
345 }

347 static RobotModelMap *getRobotModelMap(){
348 if(!robot_model_map)
349 robot_model_map = new RobotModelMap;

351 }

360template<typename T>
361struct RobotModelRegistry : RobotModelFactory{
362 RobotModelRegistry(const std::string& name) {
363 RobotModelMap::iterator it = getRobotModelMap()->find(name);
364 if(it != getRobotModelMap()->end())
365 throw std::runtime_error("Failed to register plugin with name " + name + ". A plugin with the same name is already registered");
366 getRobotModelMap()->insert(std::make_pair(name, &createT<T>));
